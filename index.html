function speak(text) { if ('speechSynthesis' in window) { synth.cancel(); const utterance = new SpeechSynthesisUtterance(text); utterance.lang = 'hi-IN'; // Politeness & Emotional Tone utterance.pitch = 1.2; // Softer female tone utterance.rate = 0.85; // Gentle & calm speed // Filter Female Voice from Available Browser Voices const voices = synth.getVoices(); const femaleVoice = voices.find(v => v.lang.includes('hi') && (v.name.includes('Female') || v.name.includes('Google') || v.name.includes('Heera'))); if (femaleVoice) { utterance.voice = femaleVoice; } document.getElementById('voiceStatus').innerText = text; synth.speak(utterance); } }
